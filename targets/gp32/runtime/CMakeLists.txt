cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(m0boy_s3c2400 LANGUAGES C)
    include(CTest)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
    endif()
endif()

option(M0_GP32_LTO "Link time optimization for the GP32 runtime" ON)
set(M0_GP32_LTO_JOBS "" CACHE STRING "Parallel link time optimization jobs (memory bound)")
set(M0_GP32_PGO "off" CACHE STRING "Profile guided optimization: off, generate or use")
set_property(CACHE M0_GP32_PGO PROPERTY STRINGS off generate use)
set(M0_GP32_PGO_DIR "${CMAKE_CURRENT_BINARY_DIR}/pgo" CACHE PATH "Profile directory")
if(M0_GP32_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT M0_GP32_HAS_LTO LANGUAGES C)
    if(M0_GP32_HAS_LTO)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
        if(M0_GP32_LTO_JOBS AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
            add_link_options(-flto=${M0_GP32_LTO_JOBS})
        endif()
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    if(M0_GP32_PGO STREQUAL "generate")
        add_compile_options(-fprofile-generate=${M0_GP32_PGO_DIR})
        link_libraries(-fprofile-generate=${M0_GP32_PGO_DIR})
    elseif(M0_GP32_PGO STREQUAL "use")
        if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
            add_compile_options(-fprofile-use=${M0_GP32_PGO_DIR} -fprofile-partial-training
                                -Wno-missing-profile)
        else()
            add_compile_options(-fprofile-use=${M0_GP32_PGO_DIR}/default.profdata)
        endif()
        link_libraries(-fprofile-use=${M0_GP32_PGO_DIR})
    endif()
endif()

# Nothing unwinds through C frames and unused functions are dropped.
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-ffunction-sections -fdata-sections -fno-asynchronous-unwind-tables
                        -fno-unwind-tables -fno-ident
                        -fno-stack-protector)
    if(NOT APPLE)
        add_link_options(-Wl,--gc-sections)
    endif()
endif()

# A single self contained .exe; the stack matches the usual Unix one.
if(WIN32 AND NOT MSVC)
    add_link_options(-static -Wl,--stack,8388608)
endif()

add_library(m0_s3c2400_bus STATIC s3c2400_bus.c)
target_include_directories(m0_s3c2400_bus PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(m0_s3c2400_bus PUBLIC c_std_99)

include(CheckCSourceCompiles)
check_c_source_compiles("
#if !defined(__x86_64__) && !defined(__i386__) && !defined(_M_X64) && !defined(_M_IX86)
#error not a PC
#endif
int main(void) { return 0; }" M0_GP32_HOST_IS_PC)
option(M0_GP32_PC_TOOLS "Screen shots and coverage maps of the missing code (PC)"
       ${M0_GP32_HOST_IS_PC})
if(M0_GP32_PC_TOOLS)
    add_compile_definitions(M0_GP32_PC_TOOLS=1)
else()
    add_compile_definitions(M0_GP32_PC_TOOLS=0)
endif()

option(M0_GP32_RAM_PROFILE "Count the uses of each 16 KiB page of the guest's RAM" OFF)
if(M0_GP32_RAM_PROFILE)
    add_compile_definitions(M0_GP32_RAM_PROFILE)
endif()

set(M0_GP32_AOT "" CACHE FILEPATH "Recompiled application generated by recompile.py")
if(M0_GP32_AOT)
    # The generated files include arm920t.c to specialize the interpreter.
    string(REGEX REPLACE "\\.c$" "" M0_GP32_AOT_STEM "${M0_GP32_AOT}")
    file(GLOB M0_GP32_AOT_PARTS CONFIGURE_DEPENDS "${M0_GP32_AOT_STEM}.part[0-9][0-9][0-9].c")
    add_library(m0_arm920t STATIC ${M0_GP32_AOT} ${M0_GP32_AOT_PARTS})
    target_include_directories(m0_arm920t PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(m0_arm920t PUBLIC M0_GP32_AOT)
else()
    add_library(m0_arm920t STATIC arm920t.c)
endif()
target_link_libraries(m0_arm920t PUBLIC m0_s3c2400_bus)

add_library(m0_s3c2400 STATIC s3c2400.c gp32_smc.c gp32_exec.c gp32_hle.c)
target_link_libraries(m0_s3c2400 PUBLIC m0_arm920t)
# Every program of this build runs one machine.
target_compile_definitions(m0_s3c2400 PRIVATE M0_GP32_ONE_MACHINE)

if(UNIX OR MINGW)
    if(WIN32)
        set(ZLIB_USE_STATIC_LIBS ON)
    endif()
    add_library(m0_gp32_host STATIC gp32_host.c gp32_state.c)
    target_link_libraries(m0_gp32_host PUBLIC m0_s3c2400 m)
    if(M0_GP32_PC_TOOLS)
        find_package(ZLIB REQUIRED) # screen shots carrying the machine state
        target_sources(m0_gp32_host PRIVATE gp32_shot.c)
        target_link_libraries(m0_gp32_host PUBLIC ZLIB::ZLIB)
    endif()
    target_compile_definitions(m0_gp32_host PRIVATE
        GP32_SYNC_CARD="${CMAKE_CURRENT_SOURCE_DIR}/../sync_card.py")
    add_executable(m0_gp32_boot gp32_boot.c)
    target_link_libraries(m0_gp32_boot PRIVATE m0_gp32_host)
    add_executable(m0_gp32_bench gp32_bench.c)
    target_link_libraries(m0_gp32_bench PRIVATE m0_gp32_host)
endif()

if(BUILD_TESTS OR (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING))
    add_executable(test_s3c2400_bus test_s3c2400_bus.c)
    target_link_libraries(test_s3c2400_bus PRIVATE m0_s3c2400_bus)
    add_test(NAME s3c2400_bus COMMAND test_s3c2400_bus)

    add_executable(test_arm920t test_arm920t.c)
    target_link_libraries(test_arm920t PRIVATE m0_arm920t)
    add_test(NAME arm920t COMMAND test_arm920t)

    add_executable(test_s3c2400 test_s3c2400.c)
    target_link_libraries(test_s3c2400 PRIVATE m0_s3c2400)
    add_test(NAME s3c2400 COMMAND test_s3c2400)

    add_executable(test_resample test_resample.c)
    if(NOT MSVC)
        target_link_libraries(test_resample PRIVATE m)
    endif()
    add_test(NAME resample COMMAND test_resample)

    # The tests check with assert, which release builds would compile out.
    foreach(test test_s3c2400_bus test_arm920t test_s3c2400 test_resample)
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endforeach()
endif()

find_package(SDL2 QUIET CONFIG)
find_package(OpenGL QUIET)
set(M0_GP32_SDL SDL2::SDL2)
option(M0_GP32_STATIC_SDL "Link SDL2 statically" OFF)
if((MINGW OR M0_GP32_STATIC_SDL) AND TARGET SDL2::SDL2-static)
    set(M0_GP32_SDL SDL2::SDL2-static)
endif()
if((UNIX OR MINGW) AND TARGET ${M0_GP32_SDL} AND TARGET OpenGL::GL)
    add_executable(m0_gp32_sdl gp32_sdl.c gp32_menu.c)
    # The font of the option menu shared by the m0boy frontends.
    target_include_directories(m0_gp32_sdl PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common/platform)
    target_compile_definitions(m0_gp32_sdl PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(m0_gp32_sdl PRIVATE m0_gp32_host ${M0_GP32_SDL} OpenGL::GL m)
    # A single file: the application, and its card, inside the executable.
    set(M0_GP32_BUNDLE "" CACHE FILEPATH "Package made by bundle.py to embed")
    if(M0_GP32_BUNDLE)
        enable_language(ASM)
        configure_file(bundle.S.in bundle.S @ONLY)
        set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/bundle.S
                                    PROPERTIES OBJECT_DEPENDS ${M0_GP32_BUNDLE})
        target_sources(m0_gp32_sdl PRIVATE gp32_bundle.c gp32_lzma.c
                       ${CMAKE_CURRENT_BINARY_DIR}/bundle.S)
        target_compile_definitions(m0_gp32_sdl PRIVATE M0_GP32_BUNDLE)
    endif()
endif()
